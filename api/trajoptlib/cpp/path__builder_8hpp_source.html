42template <typename Drivetrain, typename Solution>
43class TRAJOPT_DLLEXPORT PathBuilder {
44 public:
48 void set_drivetrain(Drivetrain drivetrain) {

59 void set_bumpers(double front, double left, double right, double back) {
60 bumpers.emplace_back(trajopt::KeepOutRegion{.safety_distance = 0.01,
61 .points = {{+front, +left},
62 {-back, +left},
63 {-back, -right},
64 {+front, -right}}});
65 }

70 std::vector<KeepOutRegion>& get_bumpers() { return bumpers; }

77 void set_control_interval_counts(std::vector<size_t>&& counts) {
78 control_interval_counts = std::move(counts);

84 const std::vector<size_t>& get_control_interval_counts() const {

92 void wpt_initial_guess_point(size_t wpt_index, const Pose2d& pose_guess) {
93 new_wpts(wpt_index);
94 initial_guess_points.at(wpt_index).back() = pose_guess;
95 }
96
106 void sgmt_initial_guess_points(size_t from_index,
107 const std::vector<Pose2d>& sgmt_pose_guess) {
108 new_wpts(from_index + 1);
109 std::vector<Pose2d>& to_initial_guess_points =
110 initial_guess_points.at(from_index + 1);
111 to_initial_guess_points.insert(to_initial_guess_points.begin(),
112 sgmt_pose_guess.begin(),
113 sgmt_pose_guess.end());
114 }

125 void pose_wpt(size_t index, double x, double y, double heading) {
126 wpt_constraint(index, PoseEqualityConstraint{x, y, heading});
127 wpt_initial_guess_point(index, {x, y, {heading}});
128 }
129
139 void translation_wpt(size_t index, double x, double y,
140 double heading_guess = 0.0) {
141 wpt_constraint(index, TranslationEqualityConstraint{x, y});
142 wpt_initial_guess_point(index, {x, y, {heading_guess}});
143 }
144
149 void wpt_constraint(size_t index, const Constraint& constraint) {
150 new_wpts(index);
151 path.waypoints.at(index).waypoint_constraints.push_back(constraint);
152 }
153
159 void sgmt_constraint(size_t from_index, size_t to_index,
160 const Constraint& constraint) {
161 assert(from_index < to_index);
162
163 new_wpts(to_index);
164 path.waypoints.at(from_index).waypoint_constraints.push_back(constraint);
165 for (size_t index = from_index + 1; index <= to_index; ++index) {
166 path.waypoints.at(index).waypoint_constraints.push_back(constraint);
167 path.waypoints.at(index).segment_constraints.push_back(constraint);
168 }
169 }
170
178 void add_callback(
179 const std::function<void(const Solution& solution, int64_t handle)>
180 callback) {
181 path.callbacks.push_back(callback);
182 }
183
187 Path<Drivetrain, Solution>& get_path() { return path; }
188
193 Solution calculate_linear_initial_guess() const {
194 return generate_linear_initial_guess<Solution>(initial_guess_points,
195 control_interval_counts);
196 }
197
202 Solution calculate_spline_initial_guess() const {
203 return generate_spline_initial_guess<Solution>(initial_guess_points,
204 control_interval_counts);
205 }
206
207 protected:
209 Path<Drivetrain, Solution> path;
210
212 std::vector<KeepOutRegion> bumpers;
213
215 std::vector<std::vector<Pose2d>> initial_guess_points;
216
218 std::vector<size_t> control_interval_counts;
219
223 void new_wpts(size_t final_index) {
224 if (final_index < path.waypoints.size()) {
225 return;
226 }
227
228 for (size_t i = path.waypoints.size(); i <= final_index; ++i) {
229 path.waypoints.emplace_back();
230 initial_guess_points.emplace_back(std::vector<Pose2d>{{0.0, 0.0, {0.0}}});
231 if (i != 0) {
232 control_interval_counts.push_back(40);
233 }
234 }
235 }
236};
